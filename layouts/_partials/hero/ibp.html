{{- /* 4. Indian-buffet-style activity matrix: 8 users (rows, sorted by first visit) × 54 days (columns). New users arrive quickly at first and then more slowly, which gives the left-ordered staircase, traced by an accent line. A grey strip runs from each user's first day; the first visit is solid, repeat visits lighter and sparse. Repeat visits are a deterministic hash, so the build is reproducible. */ -}} {{- $first := slice 0 1 3 6 10 16 25 37 }}{{ $rate := slice 24 16 28 18 26 20 30 22 }} {{- $u := 12 }}{{ $c := 9 }}{{ $x0 := 7 }}{{ $y0 := 5 }}{{ $n := 54 }} {{- $strip := "" }}{{ $new := "" }}{{ $rep := "" }}{{ $stair := printf "M%d %d" $x0 (sub $y0 2) }} {{- range $i, $f := $first }}{{ $y := add (mul $i $u) $y0 }}{{ $x := add (mul $f $u) $x0 }}{{ $p := index $rate $i }} {{- $w := sub (mul (sub $n $f) $u) (sub $u $c) }} {{- $strip = print $strip (printf "M%d %dh%dv%dh-%dz" $x $y $w $c $w) }} {{- $new = print $new (printf "M%d %dh%dv%dh-%dz" $x $y $c $c $c) }} {{- $stair = print $stair (printf "H%dV%d" (sub $x 2) (add $y (add $c 2))) }} {{- range $d := seq (add $f 1) (sub $n 1) }} {{- if lt (mod (hash.FNV32a (printf "ibp-%d-%d" $i $d)) 100) $p }}{{ $rep = print $rep (printf "M%d %dh%dv%dh-%dz" (add (mul $d $u) $x0) $y $c $c $c) }}{{ end }} {{- end }}{{ end -}}
